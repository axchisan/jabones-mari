## 7. Estructura de datos (cómo se guarda cada producto)

```ts
type Producto = {
  id: string
  slug: string                 // "menta-romero"
  nombre: string               // "Menta y Romero"
  claim: string                // "Despierta tu piel"
  descripcion: string          // texto largo
  modoDeUso: string
  advertencia: string | null   // "Contiene cúrcuma: hacer prueba..."
  ingredientes: string[]
  beneficios: string[]         // 6 items
  tipoDePiel: string[]         // ["Mixta", "Grasa"]
  uso: string[]                // ["Facial", "Corporal"]
  aroma: string
  colorMarca: string           // "#7E9A7C"
  imagenes: { url: string; alt: string }[]
  destacado: boolean
  activo: boolean              // permite ocultarlo sin borrarlo
  orden: number
  variantes: Variante[]
}

type Variante = {
  id: string
  tamano: 'grande' | 'pequeno'
  precio: number               // 7500 | 5000, en pesos enteros
  pesoGramos: number | null
  molde: string                // "Corazón, óvalo o flor (según disponibilidad)"
  disponible: boolean          // botón "Agotado" sin borrar el producto
  sku: string                  // "MENTA-G"
}
```
